cmake_minimum_required(VERSION 3.16)
project(FireflyVoiceAssistantLinux VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    message(FATAL_ERROR "FireflyVoiceAssistantLinux requires Linux/aarch64")
endif()
add_library(firefly_core src/CommandEngine.cpp src/Action.cpp src/ActionValidator.cpp src/Text.cpp src/DeterministicParser.cpp src/SafetyGate.cpp)
target_include_directories(firefly_core PUBLIC include)
target_compile_features(firefly_core PUBLIC cxx_std_17)
set_target_properties(firefly_core PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(firefly_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(firefly_tests tests/CoreTests.cpp)
    target_link_libraries(firefly_tests PRIVATE firefly_core)
    target_compile_options(firefly_tests PRIVATE -Wall -Wextra -Wpedantic)
    foreach(component Action ActionValidator DeterministicParser SafetyGate)
        add_test(NAME ${component} COMMAND firefly_tests ${component})
    endforeach()
endif()

if(BUILD_TESTING)
    add_executable(command_engine_tests tests/CommandEngineTests.cpp)
    target_link_libraries(command_engine_tests PRIVATE firefly_core)
    target_compile_options(command_engine_tests PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME CommandEngine COMMAND command_engine_tests)
endif()

option(FIREFLY_BUILD_ASSISTANT "Build the real RKLLM CLI (unit tests need no NPU)" ON)
set(RKLLM_ROOT "" CACHE PATH "RKLLM root containing rkllm.h and lib/librkllmrt.so")
if(FIREFLY_BUILD_ASSISTANT)
    find_path(RKLLM_INCLUDE_DIR rkllm.h HINTS "${RKLLM_ROOT}")
    find_library(RKLLM_LIBRARY NAMES rkllmrt HINTS "${RKLLM_ROOT}/lib")
    if(NOT RKLLM_INCLUDE_DIR OR NOT RKLLM_LIBRARY)
        message(FATAL_ERROR "Set RKLLM_ROOT to the directory containing rkllm.h and lib/librkllmrt.so, or disable FIREFLY_BUILD_ASSISTANT")
    endif()
    find_package(Threads REQUIRED)
    add_library(firefly_rkllm src/RkllmEngine.cpp)
    target_include_directories(firefly_rkllm PRIVATE "${RKLLM_INCLUDE_DIR}")
    target_link_libraries(firefly_rkllm PUBLIC firefly_core PRIVATE "${RKLLM_LIBRARY}" Threads::Threads)
    target_compile_options(firefly_rkllm PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(firefly-assistant src/main.cpp)
    target_link_libraries(firefly-assistant PRIVATE firefly_rkllm)
    target_compile_options(firefly-assistant PRIVATE -Wall -Wextra -Wpedantic)
endif()
