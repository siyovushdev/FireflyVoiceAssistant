if(FIREFLY_WAKE_MODEL_REFERENCE)
    if(NOT FIREFLY_WAKE_MODEL_REFERENCE MATCHES "^/opt/firefly/voice-assistant/releases/[A-Za-z0-9][A-Za-z0-9._+-]*/share/firefly/wake/amina-v3.tflite$")
        message(FATAL_ERROR "Amina reference must name a fixed installed release")
    endif()

    if(NOT wake_real STREQUAL FIREFLY_WAKE_MODEL_REFERENCE OR IS_SYMLINK "${FIREFLY_WAKE_MODEL_REFERENCE}")
        message(FATAL_ERROR "Amina reference must be a canonical regular file")
    endif()
    file(SHA256 "${wake_real}" wake_sha)
    if(NOT wake_sha STREQUAL "bc42cb869d2aace69c5bdf9333c502b96dedfa85df97676be85f5c289dc057d1")
        message(FATAL_ERROR "Amina reference differs from the frozen model")
    endif()
    file(CREATE_LINK "${wake_real}" "${PROJECT_BINARY_DIR}/share/firefly/wake/amina-v3.tflite" SYMBOLIC)
