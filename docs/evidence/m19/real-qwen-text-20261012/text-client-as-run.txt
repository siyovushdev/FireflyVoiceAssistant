import copy,json,time,urllib.error,urllib.request
from pathlib import Path
ROOT=Path(__file__).resolve().parent
E=ROOT/'evidence';URL='http://127.0.0.1:18080';HEAD='8a6eb084dba4aecb19af2894f5b26c6cf3ca69e0'
report={'scope':'REAL QWEN candidate TEXT ONLY. Prior reviewed genuine semantic forms reused; not new AI Setup or acoustic acceptance.','cases':[]}

def heartbeat():(ROOT/'agent-heartbeat').touch()
def save():(E/'real-text-report.json').write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n')
def api(path,body=None,method=None):
    heartbeat()
    request=urllib.request.Request(URL+path,data=None if body is None else json.dumps(body,ensure_ascii=False).encode(),method=method,headers={'Content-Type':'application/json','X-Firefly-Admin':'1'})
    try:
        with urllib.request.urlopen(request,timeout=35) as response:status,value=response.status,json.load(response)
    except urllib.error.HTTPError as error:status,value=error.code,json.load(error)
    assert status==200,(path,status,value)
    return value

def requests():
    p=E/'http.jsonl';return [] if not p.exists() else [json.loads(line) for line in p.read_text().splitlines() if line]

def state():return next(d for d in api('/api/devices/state')['devices'] if d['targetId']=='device_lampa')['properties']['power']
def test(text,on=None,route=None):
    count=len(requests());history=len(api('/api/executions')['executions'])
    response=api('/api/test-command',{'text':text,'execute':True,'production':True})
    if response.get('raw_rkllm') is not None:
        report.setdefault('all_fallback_calls',[]).append({'input':text,'raw_output':response['raw_rkllm'],'metrics':response['llm'],'safety':response['safety'],'result':response['result'],'context':response.get('llm_context')})
    delta=requests()[count:];row={'input':text,'expected_on':on,'response':response,'command_requests':delta};report['cases'].append(row);save()
    if on is None:
        assert response['result']=={'unsupported':True} and response['execution'] is None,response
        assert delta==[] and len(api('/api/executions')['executions'])==history,row
    else:
        action='action_lampa_vklyuchit' if on else 'action_lampa_vyklyuchit'
        assert response['result']=={'targetId':'device_lampa','actionId':action,'arguments':{}},response
        assert response['language_route']==route and response['safety']=='PASS' and response['execution']['success'],response
        assert len(delta)==1 and delta[0]['method']=='POST' and delta[0]['path']=='/relay/1' and json.loads(delta[0]['body'])=={'on':on},row
        assert state()['knowledge']=='predicted'
        poll=api('/api/devices/poll',{'target_id':'device_lampa','expected_hash':api('/api/config/active')['active']['hash']})
        confirmed=state();row.update(poll=poll,confirmed=confirmed,requests_after_poll=requests()[count:]);save()
        assert confirmed['knowledge']=='confirmed' and confirmed['value']==on,row
        assert len(row['requests_after_poll'])==2 and row['requests_after_poll'][1]['method']=='GET',row
        h=api('/api/executions')['executions'];assert len(h)==history+1 and h[-1]['source']=='admin-production',h
        if route=='LLM_FALLBACK':
            assert response.get('raw_rkllm') is not None,'No real model result'
            assert response['route']=='rkllm' and response['llm']['output_tokens']>0 and response['llm']['generation_ms']>0,('No actual inference metrics',response)
            report.setdefault('real_model_calls',[]).append({'input':text,'raw_output':response['raw_rkllm'],'metrics':response['llm'],'context':response.get('llm_context')})
        else:
            assert response['route']=='deterministic' and response['llm']['output_tokens']==0,response
    row['pass']=True;save()
